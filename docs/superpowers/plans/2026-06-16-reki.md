# Reki Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Reki, a Node.js server + static web page that lets a phone on the LAN send media-key and text-typing commands to the Windows PC it runs on.

**Architecture:** Express server exposes `/media/:action` and `/type` routes; a thin `lib/input.js` module wraps `@nut-tree-fork/nut-js` calls so routes stay testable without driving real keyboard hardware in tests. A static `public/` page provides buttons and a text box that call the routes via `fetch`.

**Tech Stack:** Node.js, Express, `@nut-tree-fork/nut-js`, Jest + Supertest for tests, plain HTML/CSS/JS frontend (no build step).

// lib/input.js
const { keyboard, Key } = require('@nut-tree-fork/nut-js');

const ACTION_KEYS = {
  'play-pause': Key.AudioPlay,
  next: Key.AudioNext,
  prev: Key.AudioPrev,
  'volume-up': Key.AudioVolUp,
  'volume-down': Key.AudioVolDown,
  mute: Key.AudioMute,
};

async function mediaAction(action) {
  const key = ACTION_KEYS[action];
  if (!key) {
    throw new Error(`Unknown media action: ${action}`);
  }
  await keyboard.pressKey(key);
  await keyboard.releaseKey(key);
}

async function typeText(text) {
  await keyboard.type(text);
}

module.exports = { mediaAction, typeText };
```

- [ ] **Step 1: Write server.js**

// server.js
const os = require('os');
const path = require('path');
const express = require('express');
const mediaRouter = require('./routes/media');
const typeRouter = require('./routes/type');

const PORT = 5757;

function getLanIp() {
  const interfaces = os.networkInterfaces();
  for (const name of Object.keys(interfaces)) {
    for (const iface of interfaces[name]) {
      if (iface.family === 'IPv4' && !iface.internal) {
        return iface.address;
      }
    }
  }
  return 'localhost';
}

const app = express();
app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));
app.use('/media', mediaRouter);
app.use('/type', typeRouter);

app.listen(PORT, '0.0.0.0', () => {
  const ip = getLanIp();
  console.log(`Reki running. Open this on your phone: http://${ip}:${PORT}`);
});
```

- [ ] **Step 3: Write public/app.js**

```js
const toast = document.getElementById('toast');
let toastTimer = null;

function showError(message) {
  toast.textContent = message;
  toast.classList.remove('hidden');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => toast.classList.add('hidden'), 3000);
}

async function postJson(url, body) {
  const res = await fetch(url, {
    method: 'POST',
    headers: body ? { 'Content-Type': 'application/json' } : undefined,
    body: body ? JSON.stringify(body) : undefined,
  });
  if (!res.ok) {
    const data = await res.json().catch(() => ({ error: 'Request failed' }));
    throw new Error(data.error || 'Request failed');
  }
}

document.querySelectorAll('.media-grid button').forEach((button) => {
  button.addEventListener('click', async () => {
    try {
      await postJson(`/media/${button.dataset.action}`);
    } catch (err) {
      showError(err.message);
    }
  });
});

const textInput = document.getElementById('text-input');
document.getElementById('send-text').addEventListener('click', async () => {
  const text = textInput.value;
  if (!text) return;
  try {
    await postJson('/type', { text });
    textInput.value = '';
  } catch (err) {
    showError(err.message);
  }
});
```
